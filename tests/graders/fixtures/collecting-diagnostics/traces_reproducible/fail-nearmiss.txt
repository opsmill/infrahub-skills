```bash
curl -X PUT "$INFRAHUB_ADDRESS/api/troubleshooting/trace-sample-ratio" -d '{"ratio": 1.0, "duration_seconds": 1800}'
curl "$INFRAHUB_ADDRESS/api/troubleshooting/trace-sample-ratio"
infrahub-collect create --include-traces --traces-since 30m
```
